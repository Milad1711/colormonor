    87→    87→function bezierArc(a: THREE.Vector3, b: THREE.Vector3, apex: number, t: number, out: THREE.Vector3): THREE.Vector3 {
    88→    88→  const midX = (a.x + b.x) / 2;
    89→    89→  const midY = Math.max(a.y, b.y) + apex;
    90→    90→  const midZ = (a.z + b.z) / 2;
    91→    91→  const u = 1 - t;
    92→    92→  out.set(
    93→    93→    u * u * a.x + 2 * u * t * midX + t * t * b.x,
    94→    94→    u * u * a.y + 2 * u * t * midY + t * t * b.y,
    95→    95→    u * u * a.z + 2 * u * t * midZ + t * t * b.z,
    96→    96→  );
    97→    97→  return out;
    98→    98→}

   462→   462→      scaleFxRef.current = scaleFxRef.current.filter((fx) => {

   509→   509→          node.ring.scale.setScalar(1);

   526→   526→          pos.setXYZ(

   529→   529→            pos.getY(i) + b.vels[i * 3 + 1] * dt,
   530→   530→            pos.getZ(i) + b.vels[i * 3 + 2] * dt,
   531→   531→          );
   532→   532→          b.vels[i * 3 + 1] -= 2.6 * dt;
   533→   533→        }
   534→   534→        pos.needsUpdate = true;
   535→   535→        b.mat.opacity = 1 - age / b.life;
   536→   536→        return true;
   537→   537→      });
   538→   538→
   539→   539→      // droplet arcs
   540→   540→      dropsRef.current = dropsRef.current.filter((d) => {
   541→   541→        const age = now - d.born;
   542→   542→        if (age > d.dur) {
   543→   543→          d.points.removeFromParent();
   544→   544→          d.points.geometry.dispose();
   545→   545→          d.mat.dispose();
   546→   546→          return false;
   547→   547→        }
   548→   548→        const pos = d.points.geometry.attributes.position as THREE.BufferAttribute;
   549→   549→        for (let i = 0; i < d.n; i++) {
   550→   550→          const local = clamp01((age - i * (d.dur / d.n) * 0.5) / (d.dur * 0.55));
   551→   551→          if (local <= 0 || local >= 1) {
   552→   552→            pos.setXYZ(i, d.src.x, -10, d.src.z);
   553→   553→          } else {
   554→   554→            bezierArc(d.src, d.dst, 0.55, local, V3);

   557→   557→        }
   558→   558→        pos.needsUpdate = true;
   559→   559→        return true;
   560→   560→      });
   561→   561→
   562→   562→      // ambient dust drift (wraps around the floating board)
   563→   563→      if (dustRef.current) {
   564→   564→        const d = dustRef.current;
   565→   565→        const pos = d.points.geometry.attributes.position as THREE.BufferAttribute;
   566→   566→        for (let i = 0; i < pos.count; i++) {
   567→   567→          let x = pos.getX(i) + d.vels[i * 3] * dt;
   568→   568→          let y = pos.getY(i) + d.vels[i * 3 + 1] * dt;
   569→   569→          let z = pos.getZ(i) + d.vels[i * 3 + 2] * dt;
   570→   570→          if (y > d.bounds.y * 1.45) y = -0.5;
   571→   571→          if (x > d.bounds.x) x = -d.bounds.x;
   572→   572→          else if (x < -d.bounds.x) x = d.bounds.x;
   573→   573→          if (z > d.bounds.z) z = -d.bounds.z;
   574→   574→          else if (z < -d.bounds.z) z = d.bounds.z;
   575→   575→          pos.setXYZ(i, x, y, z);
   576→   576→        }
   577→   577→        pos.needsUpdate = true;
   578→   578→      }
   579→   579→
   580→   580→      // gentle camera sway
   581→   581→      camera.position.x = Math.sin(elapsed * 0.4) * 0.14;
   582→   582→      camera.lookAt(lookTargetV);
   583→   583→
   584→   584→      renderer.render(scene, camera);
   585→   585→    };

   587→   587→    /* ------------------------------ imperative pour ------------------------------ */
   588→   588→    const tweenMs = (ms: number, onU: (p: number, tMs: number) => void) =>
   589→   589→      new Promise<void>((resolve) => {
   590→   590→        const t0 = performance.now();
   591→   591→        const step = () => {
   592→   592→          const tMs = performance.now() - t0;
   593→   593→          const p = clamp01(tMs / ms);
   594→   594→          onU(p, tMs);
   595→   595→          if (p < 1) requestAnimationFrame(step);
   596→   596→          else resolve();
   597→   597→        };
   598→   598→        requestAnimationFrame(step);
   599→   599→      });
   600→   600→
   601→   601→    pourImplRef.current = async (from, to, count, color) => {
   602→   602→        const nodeF = nodes[from];
   603→   603→        const nodeT = nodes[to];
   604→   604→        if (!nodeF || !nodeT) return;
   605→   605→        pouringRef.current = from;
   606→   606→        const basePos = nodeF.group.position.clone();
   607→   607→        const liftY = 1.25;
   608→   608→        const destMouthTop = TUBE_H + 0.72;
   609→   609→
   610→   610→        // --- phase A: fly out ---
   611→   611→        await tweenMs(FLY_OUT, (p) => {
   612→   612→          const e = 1 - Math.pow(1 - p, 3);
   613→   613→          nodeF.group.position.x = THREE.MathUtils.lerp(basePos.x, nodeT.baseX, e);
   614→   614→          nodeF.group.position.z = THREE.MathUtils.lerp(basePos.z, nodeT.baseZ, e);
   615→   615→          nodeF.group.position.y = basePos.y + liftY * Math.sin(p * Math.PI * 0.5) + Math.sin(p * Math.PI) * 0.22;
   616→   616→        });
   617→   617→
   618→   618→        // --- phase B: tilt + transfer ---
   619→   619→        const dir = new THREE.Vector3(nodeT.baseX - nodeF.baseX, 0, nodeT.baseZ - nodeF.baseZ);
   620→   620→        if (dir.lengthSq() < 1e-6) dir.set(0.01, 0, 0.05);
   621→   621→        const axis = new THREE.Vector3().crossVectors(UP, dir).normalize();
   622→   622→        const pourDur = count * UNIT_MS + 110;
   623→   623→        const srcLen = tubesRef.current[from]?.length ?? nodeF.units.length;
   624→   624→        const dstLen = tubesRef.current[to]?.length ?? 0;
   625→   625→

   631→   631→          m.scale.y = 0.01;
   632→   632→          nodeT.liquid.add(m);
   633→   633→          ghosts.push(m);
   634→   634→        }
   635→   635→

   637→   637→        const mouthSrc = nodeF.group.localToWorld(new THREE.Vector3(0, TUBE_H, 0));
   638→   638→        const mouthDst = new THREE.Vector3(nodeT.baseX, destMouthTop - 0.35, nodeT.baseZ);
   639→   639→        const dN = 9;
   640→   640→        const dGeo = new THREE.BufferGeometry();

   651→   651→          // source units drain sequentially
   652→   652→          for (let k = 0; k < count; k++) {
   653→   653→            const sp = clamp01((tMs - k * UNIT_MS) / 95);
   654→   654→            const m = nodeF.units[srcLen - 1 - k];
   655→   655→            if (m) m.scale.y = Math.max(0.02, 1 - sp * 0.98);

   659→   659→            const gp = clamp01((tMs - 90 - k * UNIT_MS) / 110);
   660→   660→            ghosts[k].scale.y = Math.max(0.01, gp);
   661→   661→          }
   662→   662→        });
   663→   663→
   664→   664→        // --- phase C: untilt + fly back ---

   666→   666→          const e = p * p * (3 - 2 * p);

   669→   669→          nodeF.group.position.z = THREE.MathUtils.lerp(nodeT.baseZ, basePos.z, e);
   670→   670→          nodeF.group.position.y = THREE.MathUtils.lerp(basePos.y + liftY, basePos.y, e);
   671→   671→        });
   672→   672→        nodeF.group.quaternion.identity();
   673→   673→        nodeF.group.position.copy(basePos);
   674→   674→
   675→   675→        ghosts.forEach((m) => m.removeFromParent());
   676→   676→        pouringRef.current = null;
   677→   677→        // let the state reconciliation place the real destination units quietly
   678→   678→        quietRef.current.add(to);
   679→   679→      };
   680→   680→
