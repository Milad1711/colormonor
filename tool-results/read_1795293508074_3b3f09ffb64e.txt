    89→    89→function bezierArc(a: THREE.Vector3, b: THREE.Vector3, apex: number, t: number, out: THREE.Vector3): THREE.Vector3 {
    90→    90→  const midX = (a.x + b.x) / 2;
    91→    91→  const midY = Math.max(a.y, b.y) + apex;
    92→    92→  const midZ = (a.z + b.z) / 2;
    93→    93→  const u = 1 - t;
    94→    94→  out.set(
    95→    95→    u * u * a.x + 2 * u * t * midX + t * t * b.x,
    96→    96→    u * u * a.y + 2 * u * t * midY + t * t * b.y,
    97→    97→    u * u * a.z + 2 * u * t * midZ + t * t * b.z,
    98→    98→  );
    99→    99→  return out;
   100→   100→}

   386→   386→    function burst(idx: number) {

   394→   394→        positions[i * 3] = origin.x;
   395→   395→        positions[i * 3 + 1] = origin.y;
   396→   396→        positions[i * 3 + 2] = origin.z;
   397→   397→        const a = (i / n) * Math.PI * 2;
   398→   398→        const r = 0.8 + Math.random() * 1.4;
   399→   399→        vels[i * 3] = Math.cos(a) * r * 0.9;
   400→   400→        vels[i * 3 + 1] = 1.6 + Math.random() * 1.8;
   401→   401→        vels[i * 3 + 2] = Math.sin(a) * r * 0.55;
   402→   402→      }
   403→   403→      const geo = new THREE.BufferGeometry();
   404→   404→      geo.setAttribute("position", new THREE.BufferAttribute(positions, 3));
   405→   405→      const mat = new THREE.PointsMaterial({ color: 0xffd54f, size: 0.13, transparent: true, opacity: 1, depthWrite: false });

   530→   530→          node.ring.scale.setScalar(1);

   558→   558→      });

   564→   564→          d.points.removeFromParent();
   565→   565→          d.points.geometry.dispose();
   566→   566→          d.mat.dispose();

   570→   570→        for (let i = 0; i < d.n; i++) {

   572→   572→          if (local <= 0 || local >= 1) {
   573→   573→            pos.setXYZ(i, d.src.x, -10, d.src.z);
   574→   574→          } else {
   575→   575→            bezierArc(d.src, d.dst, 0.55, local, V3);
   576→   576→            pos.setXYZ(i, V3.x, V3.y, V3.z);

   579→   579→        pos.needsUpdate = true;
   580→   580→        return true;
   581→   581→      });
   582→   582→
   583→   583→      // ambient dust drift (wraps around the floating board)
   584→   584→      if (dustRef.current) {
   585→   585→        const d = dustRef.current;

   589→   589→          let y = pos.getY(i) + d.vels[i * 3 + 1] * dt;
   590→   590→          let z = pos.getZ(i) + d.vels[i * 3 + 2] * dt;

   608→   608→    /* ------------------------------ imperative pour ------------------------------ */

   610→   610→      new Promise<void>((resolve) => {
   611→   611→        const t0 = performance.now();

   613→   613→          const tMs = performance.now() - t0;

   616→   616→          if (p < 1) requestAnimationFrame(step);
   617→   617→          else resolve();
   618→   618→        };
   619→   619→        requestAnimationFrame(step);

   622→   622→    pourImplRef.current = async (from, to, count, color) => {
   623→   623→        const nodeF = nodes[from];
   624→   624→        const nodeT = nodes[to];
   625→   625→        if (!nodeF || !nodeT) return;
   626→   626→        pouringRef.current = from;
   627→   627→        const basePos = nodeF.group.position.clone();
   628→   628→        const liftY = 1.25;
   629→   629→        const destMouthTop = TUBE_H + 0.72;
   630→   630→
   631→   631→        // --- phase A: fly out ---
   632→   632→        await tweenMs(FLY_OUT, (p) => {
   633→   633→          const e = 1 - Math.pow(1 - p, 3);
   634→   634→          nodeF.group.position.x = THREE.MathUtils.lerp(basePos.x, nodeT.baseX, e);
   635→   635→          nodeF.group.position.z = THREE.MathUtils.lerp(basePos.z, nodeT.baseZ, e);
   636→   636→          nodeF.group.position.y = basePos.y + liftY * Math.sin(p * Math.PI * 0.5) + Math.sin(p * Math.PI) * 0.22;
   637→   637→        });
   638→   638→
   639→   639→        // --- phase B: tilt + transfer ---
   640→   640→        const dir = new THREE.Vector3(nodeT.baseX - nodeF.baseX, 0, nodeT.baseZ - nodeF.baseZ);
   641→   641→        if (dir.lengthSq() < 1e-6) dir.set(0.01, 0, 0.05);
   642→   642→        const axis = new THREE.Vector3().crossVectors(UP, dir).normalize();
   643→   643→        const pourDur = count * UNIT_MS + 110;
   644→   644→        const srcLen = tubesRef.current[from]?.length ?? nodeF.units.length;
   645→   645→        const dstLen = tubesRef.current[to]?.length ?? 0;
   646→   646→
   647→   647→        // ghosts in the destination
   648→   648→        const ghosts: THREE.Mesh[] = [];
   649→   649→        for (let k = 0; k < count; k++) {
   650→   650→          const m = new THREE.Mesh(unitGeo, liqMats[color % liqMats.length]);

   652→   652→          m.scale.y = 0.01;
   653→   653→          nodeT.liquid.add(m);
   654→   654→          ghosts.push(m);
   655→   655→        }
   656→   656→
   657→   657→        // droplet arc
   658→   658→        const mouthSrc = nodeF.group.localToWorld(new THREE.Vector3(0, TUBE_H, 0));
   659→   659→        const mouthDst = new THREE.Vector3(nodeT.baseX, destMouthTop - 0.35, nodeT.baseZ);
   660→   660→        const dN = 9;
   661→   661→        const dGeo = new THREE.BufferGeometry();
   662→   662→        dGeo.setAttribute("position", new THREE.BufferAttribute(new Float32Array(dN * 3), 3));
   663→   663→        const dMat = new THREE.PointsMaterial({ color: LIQUIDS[color % LIQUIDS.length].base, size: 0.1, transparent: true, opacity: 0.95, depthWrite: false });
   664→   664→        const dPoints = new THREE.Points(dGeo, dMat);
   665→   665→        scene.add(dPoints);
   666→   666→        dropsRef.current.push({ points: dPoints, mat: dMat, src: mouthSrc, dst: mouthDst, born: performance.now(), dur: pourDur + 60, n: dN });
   667→   667→
   668→   668→        await tweenMs(pourDur, (_p, tMs) => {
   669→   669→          // tilt in during the first 45% of the pour
   670→   670→          const tiltP = clamp01(tMs / (pourDur * 0.45));
   671→   671→          const tiltE = tiltP * tiltP * (3 - 2 * tiltP);
   672→   672→          nodeF.group.quaternion.setFromAxisAngle(axis, TILT * tiltE);
   673→   673→          // liquid glides toward the mouth as the tube tips past horizontal
   674→   674→          const slide = tiltE * 0.5;
   675→   675→          // source units drain sequentially — the bottom edge of each unit
   676→   676→          // stays glued to the stack so the column never "floats in air"
   677→   677→          for (let k = 0; k < count; k++) {
   678→   678→            const sp = clamp01((tMs - k * UNIT_MS) / 95);
   679→   679→            const idx = srcLen - 1 - k;
   680→   680→            const m = nodeF.units[idx];
   681→   681→            if (m) {
   682→   682→              const s = Math.max(0.02, 1 - sp * 0.98);
   683→   683→              m.scale.y = s;
   684→   684→              m.position.y = bottomY(idx) + slide - HALF_H * (1 - s);
   685→   685→            }
   686→   686→          }
   687→   687→          // the still-liquid units slide up with the tilt, keeping the stack
   688→   688→          for (let k = 0; k < srcLen - count; k++) {
   689→   689→            const m = nodeF.units[k];
   690→   690→            if (m) m.position.y = bottomY(k) + slide;
   691→   691→          }
   692→   692→          // destination ghosts fill sequentially, growing upward from the base
   693→   693→          for (let k = 0; k < count; k++) {
   694→   694→            const gp = clamp01((tMs - 90 - k * UNIT_MS) / 110);
   695→   695→            const s = Math.max(0.01, gp);
   696→   696→            ghosts[k].scale.y = s;
   697→   697→            ghosts[k].position.y = bottomY(dstLen + k) - HALF_H * (1 - s);
   698→   698→          }
   699→   699→        });
   700→   700→
   701→   701→        // --- phase C: untilt + fly back ---
   702→   702→        await tweenMs(FLY_BACK, (p) => {
   703→   703→          const e = p * p * (3 - 2 * p);
   704→   704→          nodeF.group.quaternion.setFromAxisAngle(axis, TILT * (1 - e));
   705→   705→          nodeF.group.position.x = THREE.MathUtils.lerp(nodeT.baseX, basePos.x, e);
   706→   706→          nodeF.group.position.z = THREE.MathUtils.lerp(nodeT.baseZ, basePos.z, e);
   707→   707→          nodeF.group.position.y = THREE.MathUtils.lerp(basePos.y + liftY, basePos.y, e);
   708→   708→        });
   709→   709→        nodeF.group.quaternion.identity();
   710→   710→        nodeF.group.position.copy(basePos);
   711→   711→        // pin every remaining source unit back onto its exact stack slot
   712→   712→        nodeF.units.forEach((m, k) => {
   713→   713→          m.position.y = bottomY(k);
   714→   714→          if (k < srcLen - count) m.scale.y = 1;
   715→   715→        });
   716→   716→
   717→   717→        ghosts.forEach((m) => m.removeFromParent());
   718→   718→        pouringRef.current = null;
   719→   719→        // let the state reconciliation place the real destination units quietly
   720→   720→        quietRef.current.add(to);
   721→   721→      };
   722→   722→
   723→   723→    buildBoard(tubesRef.current);
   724→   724→    completeRef.current = tubesRef.current.map(isTubeComplete);

   732→   732→
   733→   733→    const onVis = () => {

   764→   764→        dustRef.current.mat.dispose();

   766→   766→      }
   767→   767→      scene.clear();
   768→   768→      renderer.dispose();
   769→   769→      canvas.remove();
   770→   770→    };
   771→   771→  }, []);
   772→   772→

   774→   774→  useEffect(() => {

   778→   778→
