   387→const FA_DIGITS = ["۰", "۱", "۲", "۳", "۴", "۵", "۶", "۷", "۸", "۹"];

   389→/** Locale-correct digits: Persian numerals in fa, Western in en. */
   390→export function fmtNum(lang: Lang, n: number): string {
   391→  const s = String(n);
   392→  if (lang === "fa") return s.replace(/\d/g, (d) => FA_DIGITS[Number(d)]);
   393→  return s;
   394→}
