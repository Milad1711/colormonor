    89→function bezierArc(a: THREE.Vector3, b: THREE.Vector3, apex: number, t: number, out: THREE.Vector3): THREE.Vector3 {
    90→  const midX = (a.x + b.x) / 2;
    91→  const midY = Math.max(a.y, b.y) + apex;
    92→  const midZ = (a.z + b.z) / 2;
    93→  const u = 1 - t;
    94→  out.set(
    95→    u * u * a.x + 2 * u * t * midX + t * t * b.x,
    96→    u * u * a.y + 2 * u * t * midY + t * t * b.y,
    97→    u * u * a.z + 2 * u * t * midZ + t * t * b.z,
    98→  );
    99→  return out;
   100→}

   262→    function buildBoard(state: number[][]) {
   263→      // clear previous tubes
   264→      for (const n of nodes) {
   265→        n.group.removeFromParent();
   266→        disposeObject(n.group);
   267→      }
   268→      nodes.length = 0;
   269→      scaleFxRef.current = [];
   270→      growRef.current = [];
   271→
   272→      // portrait screens: split into two depth rows so the board uses the
   273→      // vertical space instead of shrinking into a distant single row

   275→      const twoRows = aspect < 1.0 && state.length >= 4;
   276→      const rows = twoRows
   277→        ? [Math.ceil(state.length / 2), Math.floor(state.length / 2)]

   286→        const x = (k - (countInRow - 1) / 2) * SPACING;

   292→        const glass = new THREE.Mesh(glassGeo, glassMat);
   293→        glass.position.y = TUBE_H / 2 + 0.08;
   294→        group.add(glass);
   295→

   297→        rim.rotation.x = Math.PI / 2;
   298→        rim.position.y = TUBE_H + 0.08;
   299→        group.add(rim);
   300→
   301→        const ringMat = new THREE.MeshBasicMaterial({ color: 0xf5b301, transparent: true, opacity: 0 });
   302→        const ring = new THREE.Mesh(ringGeo, ringMat);
   303→        ring.rotation.x = Math.PI / 2;
   304→        ring.position.y = 0.09;
   305→        group.add(ring);
   306→
   307→        const hit = new THREE.Mesh(hitGeo, new THREE.MeshBasicMaterial({ transparent: true, opacity: 0, depthWrite: false }));
   308→        hit.position.y = TUBE_H / 2;
   309→        hit.userData.tubeIndex = i;
   310→        group.add(hit);
   311→
   312→        const liquid = new THREE.Group();
   313→        group.add(liquid);
   314→
   315→        const node: TubeNode = { group, liquid, ring, ringMat, hit, baseX: x, baseZ: z, floatPhase: i * 0.83 + Math.abs(x) * 0.6, units: [] };

   317→        boardRoot.add(group);
   318→

   320→        units.forEach((c, k2) => {
   321→          const m = makeUnit(c, k2);
   322→          m.scale.y = 0.01;

   324→          node.liquid.add(m);
   325→          growRef.current.push({ mesh: m, p: -k2 * 0.35, k: k2 });
   326→        });
   327→      });

   331→      const height = TUBE_H + 1.0;
   332→      const depth = (rows.length - 1) * ROW_Z * 2 + 1.4;
   333→      const fit = fitBoardCamera(camera, width, height, depth, new THREE.Vector3(0, height * 0.42, 0), {
   334→        headroom: 1.35,
   335→        margin: 1.04,
   336→        lookBias: -0.05,
   337→      });

   341→    function syncTubes(state: number[][]) {
   342→      state.forEach((target, i) => {
   343→        const node = nodes[i];
   344→        if (!node) return;
   345→        const quiet = quietRef.current.has(i);
   346→        // shrink: remove extras
   347→        while (node.units.length > target.length) {
   348→          const m = node.units.pop();
   349→          if (m) m.removeFromParent();
   350→        }
   351→        // grow: add missing
   352→        while (node.units.length < target.length) {
   353→          const k = node.units.length;
   354→          const m = makeUnit(target[k], k);
   355→          if (quiet) {
   356→            m.scale.y = 1;
   357→          } else {
   358→            m.scale.y = 0.01;
   359→            growRef.current.push({ mesh: m, p: 0, k });
   360→          }
   361→          node.units.push(m);
   362→          node.liquid.add(m);
   363→        }
   364→        // rebuild if colors differ with same length
   365→        if (node.units.length === target.length) {
   366→          const mismatch = target.some((c, k) => node.units[k].material !== liqMats[c % liqMats.length]);
   367→          if (mismatch) {
   368→            node.units.forEach((m) => m.removeFromParent());
   369→            node.units = [];
   370→            target.forEach((c, k) => {
   371→              const m = makeUnit(c, k);
   372→              node.units.push(m);
   373→              node.liquid.add(m);
   374→            });
   375→          }
   376→        }
   377→        // hard guarantee: every kept unit sits EXACTLY on its stack slot so
   378→        // colors never drift or hover after a tube empties and refills
   379→        node.units.forEach((m, k) => {
   380→          m.position.y = bottomY(k);
   381→        });
   382→      });
   383→      quietRef.current.clear();
   384→    }

   386→    function burst(idx: number) {
   387→      const node = nodes[idx];
   388→      if (!node) return;
   389→      const n = 30;
   390→      const positions = new Float32Array(n * 3);
   391→      const vels = new Float32Array(n * 3);
   392→      const origin = new THREE.Vector3(node.baseX, TUBE_H * 0.62, node.baseZ);
   393→      for (let i = 0; i < n; i++) {
   394→        positions[i * 3] = origin.x;
   395→        positions[i * 3 + 1] = origin.y;
   396→        positions[i * 3 + 2] = origin.z;
   397→        const a = (i / n) * Math.PI * 2;
   398→        const r = 0.8 + Math.random() * 1.4;
   399→        vels[i * 3] = Math.cos(a) * r * 0.9;
   400→        vels[i * 3 + 1] = 1.6 + Math.random() * 1.8;
   401→        vels[i * 3 + 2] = Math.sin(a) * r * 0.55;
   402→      }
   403→      const geo = new THREE.BufferGeometry();
   404→      geo.setAttribute("position", new THREE.BufferAttribute(positions, 3));
   405→      const mat = new THREE.PointsMaterial({ color: 0xffd54f, size: 0.13, transparent: true, opacity: 1, depthWrite: false });
   406→      const points = new THREE.Points(geo, mat);
   407→      scene.add(points);
   408→      burstsRef.current.push({ points, mat, vels, born: performance.now(), life: 950 });
   409→    }

   440→      const h = el.clientHeight || 1;

   449→    /* ------------------------------ loop ------------------------------ */
   450→    const clock = new THREE.Clock();
   451→    let raf = 0;
   452→    let running = true;
   453→    let elapsed = 0;
   454→    const lookTargetV = new THREE.Vector3(0, 1.2, 0);
   455→    const prevShkRef = { current: null as number | null };
   456→
   457→    const tick = () => {
   458→      if (!running) return;
   459→      raf = requestAnimationFrame(tick);
   460→      const dt = Math.min(clock.getDelta(), 0.05);
   461→      elapsed += dt;
   462→      const now = performance.now();
   463→      const { selected: sel, hinted: hin, shakeIdx: shk } = propsRef.current;
   464→
   465→      // unit grow pop (easeOutBack) — liquid rises from the stack base,
   466→      // the bottom edge stays glued to its slot while the column grows
   467→      growRef.current = growRef.current.filter((g) => {
   468→        g.p += dt * 3.4;
   469→        const p = clamp01(g.p);
   470→        const back = 1 + 2.2 * Math.pow(p - 1, 3) + 1.2 * Math.pow(p - 1, 2);
   471→        const s = Math.max(0.01, p < 1 ? back : 1);
   472→        g.mesh.scale.y = s;
   473→        g.mesh.position.y = bottomY(g.k) - HALF_H * (1 - s);
   474→        if (g.p >= 1) {
   475→          g.mesh.scale.y = 1;
   476→          g.mesh.position.y = bottomY(g.k);
   477→          return false;
   478→        }
   479→        return true;
   480→      });
   481→
   482→      // scale fx (source shrink handled by pour tween, kept generic)
   483→      scaleFxRef.current = scaleFxRef.current.filter((fx) => {
   484→        const p = clamp01((now - fx.t0) / fx.dur);
   485→        fx.mesh.scale.y = fx.from + (fx.to - fx.from) * p;
   486→        return p < 1;
   487→      });
   488→
   489→      // per-tube visuals
   490→      nodes.forEach((node, i) => {
   491→        if (pouringRef.current !== i) {
   492→          // gentle idle bob so the tubes float in the air
   493→          const bob = Math.sin(elapsed * 1.15 + node.floatPhase) * 0.07;
   494→          const targetLift = (sel === i ? 0.42 : 0) + bob;
   495→          node.group.position.y += (targetLift - node.group.position.y) * Math.min(1, dt * 8);
   496→          // shake (only re-armed when the shake target changes)
   497→          if (shk !== prevShkRef.current) {
   498→            prevShkRef.current = shk;
   499→            if (shk === i) shakeRef.current = { idx: i, until: now + 380 };
   500→          }
   501→          if (shakeRef.current && shakeRef.current.idx === i) {
   502→            const remain = shakeRef.current.until - now;
   503→            if (remain <= 0) {
   504→              shakeRef.current = null;
   505→              node.group.position.x = node.baseX;
   506→            } else {
   507→              const decay = remain / 380;
   508→              node.group.position.x = node.baseX + Math.sin(elapsed * 55) * 0.09 * decay;
   509→            }
   510→          } else {
   511→            node.group.position.x += (node.baseX - node.group.position.x) * Math.min(1, dt * 10);
   512→          }
   513→        }
   514→
   515→        // ring state
   516→        const complete = completeRef.current[i] ?? false;

   556→        b.mat.opacity = 1 - age / b.life;
   557→        return true;

   561→      dropsRef.current = dropsRef.current.filter((d) => {
   562→        const age = now - d.born;
   563→        if (age > d.dur) {
   564→          d.points.removeFromParent();
   565→          d.points.geometry.dispose();
   566→          d.mat.dispose();
   567→          return false;
   568→        }
   569→        const pos = d.points.geometry.attributes.position as THREE.BufferAttribute;
   570→        for (let i = 0; i < d.n; i++) {

   572→          if (local <= 0 || local >= 1) {

   577→          }
   578→        }
   579→        pos.needsUpdate = true;
   580→        return true;

   589→          let y = pos.getY(i) + d.vels[i * 3 + 1] * dt;

   593→          else if (x < -d.bounds.x) x = d.bounds.x;

   595→          else if (z < -d.bounds.z) z = d.bounds.z;
   596→          pos.setXYZ(i, x, y, z);
   597→        }
   598→        pos.needsUpdate = true;
   599→      }

   608→    /* ------------------------------ imperative pour ------------------------------ */
   609→    const tweenMs = (ms: number, onU: (p: number, tMs: number) => void) =>
   610→      new Promise<void>((resolve) => {
   611→        const t0 = performance.now();
   612→        const step = () => {
   613→          const tMs = performance.now() - t0;
   614→          const p = clamp01(tMs / ms);
   615→          onU(p, tMs);
   616→          if (p < 1) requestAnimationFrame(step);
   617→          else resolve();
   618→        };
   619→        requestAnimationFrame(step);

   622→    pourImplRef.current = async (from, to, count, color) => {
   623→        const nodeF = nodes[from];
   624→        const nodeT = nodes[to];
   625→        if (!nodeF || !nodeT) return;
   626→        pouringRef.current = from;
   627→        const basePos = nodeF.group.position.clone();
   628→        const liftY = 1.25;
   629→        const destMouthTop = TUBE_H + 0.72;
   630→
   631→        // --- phase A: fly out ---
   632→        await tweenMs(FLY_OUT, (p) => {
   633→          const e = 1 - Math.pow(1 - p, 3);
   634→          nodeF.group.position.x = THREE.MathUtils.lerp(basePos.x, nodeT.baseX, e);
   635→          nodeF.group.position.z = THREE.MathUtils.lerp(basePos.z, nodeT.baseZ, e);
   636→          nodeF.group.position.y = basePos.y + liftY * Math.sin(p * Math.PI * 0.5) + Math.sin(p * Math.PI) * 0.22;
   637→        });
   638→
   639→        // --- phase B: tilt + transfer ---
   640→        const dir = new THREE.Vector3(nodeT.baseX - nodeF.baseX, 0, nodeT.baseZ - nodeF.baseZ);
   641→        if (dir.lengthSq() < 1e-6) dir.set(0.01, 0, 0.05);
   642→        const axis = new THREE.Vector3().crossVectors(UP, dir).normalize();
   643→        const pourDur = count * UNIT_MS + 110;
   644→        const srcLen = tubesRef.current[from]?.length ?? nodeF.units.length;
   645→        const dstLen = tubesRef.current[to]?.length ?? 0;
   646→
   647→        // ghosts in the destination
   648→        const ghosts: THREE.Mesh[] = [];
   649→        for (let k = 0; k < count; k++) {
   650→          const m = new THREE.Mesh(unitGeo, liqMats[color % liqMats.length]);
   651→          m.position.y = bottomY(dstLen + k);
   652→          m.scale.y = 0.01;
   653→          nodeT.liquid.add(m);
   654→          ghosts.push(m);
   655→        }
   656→
   657→        // droplet arc
   658→        const mouthSrc = nodeF.group.localToWorld(new THREE.Vector3(0, TUBE_H, 0));
   659→        const mouthDst = new THREE.Vector3(nodeT.baseX, destMouthTop - 0.35, nodeT.baseZ);
   660→        const dN = 9;
   661→        const dGeo = new THREE.BufferGeometry();
   662→        dGeo.setAttribute("position", new THREE.BufferAttribute(new Float32Array(dN * 3), 3));
   663→        const dMat = new THREE.PointsMaterial({ color: LIQUIDS[color % LIQUIDS.length].base, size: 0.1, transparent: true, opacity: 0.95, depthWrite: false });
   664→        const dPoints = new THREE.Points(dGeo, dMat);
   665→        scene.add(dPoints);
   666→        dropsRef.current.push({ points: dPoints, mat: dMat, src: mouthSrc, dst: mouthDst, born: performance.now(), dur: pourDur + 60, n: dN });
   667→
   668→        await tweenMs(pourDur, (_p, tMs) => {
   669→          // tilt in during the first 45% of the pour
   670→          const tiltP = clamp01(tMs / (pourDur * 0.45));
   671→          const tiltE = tiltP * tiltP * (3 - 2 * tiltP);
   672→          nodeF.group.quaternion.setFromAxisAngle(axis, TILT * tiltE);
   673→          // liquid glides toward the mouth as the tube tips past horizontal
   674→          const slide = tiltE * 0.5;
   675→          // source units drain sequentially — the bottom edge of each unit
   676→          // stays glued to the stack so the column never "floats in air"
   677→          for (let k = 0; k < count; k++) {
   678→            const sp = clamp01((tMs - k * UNIT_MS) / 95);
   679→            const idx = srcLen - 1 - k;
   680→            const m = nodeF.units[idx];
   681→            if (m) {
   682→              const s = Math.max(0.02, 1 - sp * 0.98);
   683→              m.scale.y = s;
   684→              m.position.y = bottomY(idx) + slide - HALF_H * (1 - s);
   685→            }
   686→          }
   687→          // the still-liquid units slide up with the tilt, keeping the stack
   688→          for (let k = 0; k < srcLen - count; k++) {
   689→            const m = nodeF.units[k];
   690→            if (m) m.position.y = bottomY(k) + slide;
   691→          }
   692→          // destination ghosts fill sequentially, growing upward from the base
   693→          for (let k = 0; k < count; k++) {
   694→            const gp = clamp01((tMs - 90 - k * UNIT_MS) / 110);
   695→            const s = Math.max(0.01, gp);
   696→            ghosts[k].scale.y = s;
   697→            ghosts[k].position.y = bottomY(dstLen + k) - HALF_H * (1 - s);
   698→          }
   699→        });
   700→
   701→        // --- phase C: untilt + fly back ---
   702→        await tweenMs(FLY_BACK, (p) => {
   703→          const e = p * p * (3 - 2 * p);
   704→          nodeF.group.quaternion.setFromAxisAngle(axis, TILT * (1 - e));
   705→          nodeF.group.position.x = THREE.MathUtils.lerp(nodeT.baseX, basePos.x, e);
   706→          nodeF.group.position.z = THREE.MathUtils.lerp(nodeT.baseZ, basePos.z, e);
   707→          nodeF.group.position.y = THREE.MathUtils.lerp(basePos.y + liftY, basePos.y, e);
   708→        });
   709→        nodeF.group.quaternion.identity();
   710→        nodeF.group.position.copy(basePos);
   711→        // pin every remaining source unit back onto its exact stack slot
   712→        nodeF.units.forEach((m, k) => {
   713→          m.position.y = bottomY(k);
   714→          if (k < srcLen - count) m.scale.y = 1;
   715→        });
   716→
   717→        ghosts.forEach((m) => m.removeFromParent());
   718→        pouringRef.current = null;
   719→        // let the state reconciliation place the real destination units quietly
   720→        quietRef.current.add(to);
   721→      };
