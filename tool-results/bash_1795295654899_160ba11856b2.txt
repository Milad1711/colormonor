     3→/**
     4→ * Tubes3D — the water-sort puzzle rendered as a real 3D scene.
     5→ * Glass tubes and colors FLOAT freely in the air (no table, no stands —
     6→ * the tubes hover with a gentle bob) on clearly separated "floating shelves":
     7→ *
     8→ *  - 1..5 tubes  → one single horizontal row (big, side by side)
     9→ *  - 6+ tubes    → two vertically stacked rows with a wide air gap, so
    10→ *                  every tube is fully visible and tappable — nothing is
    11→ *                  ever hidden behind another tube.
    12→ *
    13→ * The pour is an in-place gesture: the source tube lifts, tips towards the
    14→ * destination, a color stream arcs into its mouth, units drain / fill in
    15→ * exact stack slots, then the tube settles back. Liquids always stack from
    16→ * the tube floor with zero drift; emptied slivers are hidden instantly.
    17→ *
    18→ * Exposes the same imperative `pour()` API as the DOM fallback board.
    19→ */

    93→function bezierArc(a: THREE.Vector3, b: THREE.Vector3, apex: number, t: number, out: THREE.Vector3): THREE.Vector3 {
    94→  const midX = (a.x + b.x) / 2;
    95→  const midY = Math.max(a.y, b.y) + apex;
    96→  const midZ = (a.z + b.z) / 2;
    97→  const u = 1 - t;
    98→  out.set(
    99→    u * u * a.x + 2 * u * t * midX + t * t * b.x,
   100→    u * u * a.y + 2 * u * t * midY + t * t * b.y,
   101→    u * u * a.z + 2 * u * t * midZ + t * t * b.z,
   102→  );
   103→  return out;
   104→}

   231→    function slotBase(i: number, n: number): { x: number; y: number; rowTop: boolean } {
   232→      const { rows, twoRows } = layoutFor(n);

   236→      const k = i - startInRow;
   237→      return { x: (k - (rowCount - 1) / 2) * SPACING, y: rowTop ? RISE : 0, rowTop };

   317→    function buildBoard(state: number[][]) {
   318→      // clear previous tubes
   319→      for (const n of nodes) {
   320→        n.group.removeFromParent();
   321→        disposeObject(n.group);
   322→      }
   323→      nodes.length = 0;
   324→      growRef.current = [];

   326→      state.forEach((units, i) => {
   327→        const { x, y } = slotBase(i, state.length);
   328→
   329→        const group = new THREE.Group();
   330→        group.position.set(x, y, 0);
   331→
   332→        const glass = new THREE.Mesh(glassGeo, glassMat);
   333→        glass.position.y = TUBE_H / 2 + 0.08;
   334→        group.add(glass);
   335→
   336→        const rim = new THREE.Mesh(new THREE.TorusGeometry(TUBE_R + 0.02, 0.045, 10, 26), rimMat);
   337→        rim.rotation.x = Math.PI / 2;
   338→        rim.position.y = TUBE_H + 0.08;
   339→        group.add(rim);
   340→
   341→        const ringMat = new THREE.MeshBasicMaterial({ color: 0xf5b301, transparent: true, opacity: 0 });
   342→        const ring = new THREE.Mesh(ringGeo, ringMat);
   343→        ring.rotation.x = Math.PI / 2;
   344→        ring.position.y = 0.09;
   345→        group.add(ring);
   346→
   347→        const hit = new THREE.Mesh(hitGeo, new THREE.MeshBasicMaterial({ transparent: true, opacity: 0, depthWrite: false }));
   348→        hit.position.y = TUBE_H / 2;
   349→        hit.userData.tubeIndex = i;
   350→        group.add(hit);
   351→
   352→        const liquid = new THREE.Group();
   353→        group.add(liquid);
   354→
   355→        const node: TubeNode = { group, liquid, ring, ringMat, hit, baseX: x, baseY: y, baseZ: 0, floatPhase: i * 0.83 + Math.abs(x) * 0.6, units: [] };
   356→        nodes.push(node);
   357→        boardRoot.add(group);
   358→
   359→        // initial units rise with a small stagger
   360→        units.forEach((c, k2) => {
   361→          const m = makeUnit(c, k2);
   362→          m.scale.y = 0.01;
   363→          node.units.push(m);
   364→          node.liquid.add(m);
   365→          growRef.current.push({ mesh: m, p: -k2 * 0.35, k: k2 });
   366→        });
   367→      });
   368→
   369→      refitCamera();
   370→    }

   372→    function syncTubes(state: number[][]) {
   373→      state.forEach((target, i) => {
   374→        const node = nodes[i];
   375→        if (!node) return;
   376→        const quiet = quietRef.current.has(i);
   377→        // shrink: remove extras
   378→        while (node.units.length > target.length) {
   379→          const m = node.units.pop();
   380→          if (m) m.removeFromParent();
   381→        }
   382→        // grow: add missing
   383→        while (node.units.length < target.length) {
   384→          const k = node.units.length;
   385→          const m = makeUnit(target[k], k);
   386→          if (quiet) {
   387→            m.scale.y = 1;
   388→          } else {
   389→            m.scale.y = 0.01;
   390→            growRef.current.push({ mesh: m, p: 0, k });
   391→          }
   392→          node.units.push(m);
   393→          node.liquid.add(m);
   394→        }
   395→        // rebuild if colors differ with same length
   396→        if (node.units.length === target.length) {
   397→          const mismatch = target.some((c, k) => node.units[k].material !== liqMats[c % liqMats.length]);
   398→          if (mismatch) {
   399→            node.units.forEach((m) => m.removeFromParent());
   400→            node.units = [];
   401→            target.forEach((c, k) => {
   402→              const m = makeUnit(c, k);
   403→              node.units.push(m);
   404→              node.liquid.add(m);
   405→            });
   406→          }
   407→        }
   408→        // hard guarantee: every kept unit sits EXACTLY on its stack slot so
   409→        // colors never drift or hover after a tube empties and refills
   410→        node.units.forEach((m, k) => {
   411→          m.position.y = bottomY(k);
   412→          if (m.scale.y < 0.99 && m.scale.y > 0.02 && !growRef.current.some((g) => g.mesh === m)) {
   413→            // restore any unit left mid-shrink by an interrupted pour
   414→            m.scale.y = 1;
   415→          }
   416→          m.visible = true;
   417→        });
   418→      });
   419→      quietRef.current.clear();
   420→    }

   422→    function burst(idx: number) {
   423→      const node = nodes[idx];
   424→      if (!node) return;
   425→      const n = 30;
   426→      const positions = new Float32Array(n * 3);
   427→      const vels = new Float32Array(n * 3);
   428→      const origin = new THREE.Vector3(node.baseX, node.baseY + TUBE_H * 0.62, node.baseZ);
   429→      for (let i = 0; i < n; i++) {
   430→        positions[i * 3] = origin.x;
   431→        positions[i * 3 + 1] = origin.y;
   432→        positions[i * 3 + 2] = origin.z;
   433→        const a = (i / n) * Math.PI * 2;
   434→        const r = 0.8 + Math.random() * 1.4;
   435→        vels[i * 3] = Math.cos(a) * r * 0.9;
   436→        vels[i * 3 + 1] = 1.6 + Math.random() * 1.8;
   437→        vels[i * 3 + 2] = Math.sin(a) * r * 0.55;
   438→      }
   439→      const geo = new THREE.BufferGeometry();
   440→      geo.setAttribute("position", new THREE.BufferAttribute(positions, 3));
   441→      const mat = new THREE.PointsMaterial({ color: 0xffd54f, size: 0.13, transparent: true, opacity: 1, depthWrite: false });
   442→      const points = new THREE.Points(geo, mat);
   443→      scene.add(points);
   444→      burstsRef.current.push({ points, mat, vels, born: performance.now(), life: 950 });
   445→    }

   493→    const tick = () => {

   501→      // unit grow pop (easeOutBack) — liquid rises from the stack base,

   504→        g.p += dt * 3.4;
   505→        const p = clamp01(g.p);
   506→        const back = 1 + 2.2 * Math.pow(p - 1, 3) + 1.2 * Math.pow(p - 1, 2);
   507→        const s = Math.max(0.01, p < 1 ? back : 1);
   508→        g.mesh.scale.y = s;
   509→        g.mesh.position.y = bottomY(g.k) - HALF_H * (1 - s);
   510→        if (g.p >= 1) {
   511→          g.mesh.scale.y = 1;
   512→          g.mesh.position.y = bottomY(g.k);

   514→        }

   607→        }
   608→        pos.needsUpdate = true;
   609→        return true;

   622→          else if (x < -d.bounds.x) x = d.bounds.x;

   624→          else if (z < -d.bounds.z) z = d.bounds.z;
   625→          pos.setXYZ(i, x, y, z);
   626→        }
   627→        pos.needsUpdate = true;
   628→      }

   632→      camera.lookAt(lookTargetV);

   637→    /* ------------------------------ imperative pour ------------------------------ */
   638→    const tweenMs = (ms: number, onU: (p: number, tMs: number) => void) =>
   639→      new Promise<void>((resolve) => {
   640→        const t0 = performance.now();
   641→        const step = () => {
   642→          const tMs = performance.now() - t0;
   643→          const p = clamp01(tMs / ms);
   644→          onU(p, tMs);
   645→          if (p < 1) requestAnimationFrame(step);
   646→          else resolve();
   647→        };

   651→    pourImplRef.current = async (from, to, count, color) => {
   652→        const nodeF = nodes[from];
   653→        const nodeT = nodes[to];
   654→        if (!nodeF || !nodeT) return;
   655→        pouringRef.current = from;
   656→        const basePos = nodeF.group.position.clone();
   657→        const mouthRestY = nodeF.baseY + TUBE_H;
   658→
   659→        /* --- phase A: rise straight up (no flying across the board) --- */
   660→        await tweenMs(FLY_OUT, (p) => {
   661→          const e = 1 - Math.pow(1 - p, 3);
   662→          nodeF.group.position.y = basePos.y + POUR_LIFT * e;
   663→        });
   664→
   665→        /* --- phase B: tip towards the destination + transfer --- */
   666→        let dx = nodeT.baseX - nodeF.baseX;
   667→        if (Math.abs(dx) < 0.12) dx = dx >= 0 ? 0.35 : -0.35; // vertical pour → tip sideways
   668→        const dir = new THREE.Vector3(dx, 0, nodeT.baseZ - nodeF.baseZ);
   669→        const axis = new THREE.Vector3().crossVectors(UP, dir).normalize();
   670→        const pourDur = count * UNIT_MS + 110;
   671→        const srcLen = tubesRef.current[from]?.length ?? nodeF.units.length;
   672→        const dstLen = tubesRef.current[to]?.length ?? 0;
   673→
   674→        // ghosts in the destination — grow from its exact stack slots
   675→        const ghosts: THREE.Mesh[] = [];

   677→          const m = new THREE.Mesh(unitGeo, liqMats[color % liqMats.length]);
   678→          m.position.y = bottomY(dstLen + k);
   679→          m.scale.y = 0.01;
   680→          nodeT.liquid.add(m);
   681→          ghosts.push(m);
   682→        }
   683→
   684→        // color stream: from the tipped mouth to the destination mouth
   685→        const mouthLocal = new THREE.Vector3(0, TUBE_H, 0).applyAxisAngle(axis, TILT * 0.8);
   686→        const mouthSrc = mouthLocal.add(nodeF.group.position);
   687→        const mouthDst = new THREE.Vector3(nodeT.baseX, nodeT.baseY + TUBE_H + 0.04, nodeT.baseZ);
   688→        const dN = 9;
   689→        const dGeo = new THREE.BufferGeometry();
   690→        dGeo.setAttribute("position", new THREE.BufferAttribute(new Float32Array(dN * 3), 3));
   691→        const dMat = new THREE.PointsMaterial({ color: LIQUIDS[color % LIQUIDS.length].base, size: 0.1, transparent: true, opacity: 0.95, depthWrite: false });
   692→        const dPoints = new THREE.Points(dGeo, dMat);
   693→        scene.add(dPoints);
   694→        dropsRef.current.push({ points: dPoints, mat: dMat, src: mouthSrc, dst: mouthDst, born: performance.now(), dur: pourDur + 60, n: dN });
   695→
   696→        await tweenMs(pourDur, (_p, tMs) => {
   697→          // tip in during the first 45% of the pour
   698→          const tiltP = clamp01(tMs / (pourDur * 0.45));
   699→          const tiltE = tiltP * tiltP * (3 - 2 * tiltP);
   700→          nodeF.group.quaternion.setFromAxisAngle(axis, TILT * tiltE);
   701→          // liquid glides towards the mouth as the tube tips past horizontal
   702→          const slide = tiltE * 0.55;
   703→          // source units drain sequentially — the bottom edge of each unit
   704→          // stays glued to its stack slot, and drained slivers vanish
   705→          for (let k = 0; k < count; k++) {
   706→            const sp = clamp01((tMs - k * UNIT_MS) / 95);
   707→            const idx = srcLen - 1 - k;
   708→            const m = nodeF.units[idx];
   709→            if (m) {
   710→              const s = Math.max(0.02, 1 - sp * 0.98);
   711→              m.scale.y = s;
   712→              m.position.y = bottomY(idx) + slide - HALF_H * (1 - s);
   713→              m.visible = s > 0.05;
   714→            }
   715→          }
   716→          // the still-liquid units ride up with the tilt, keeping the stack
   717→          for (let k = 0; k < srcLen - count; k++) {
   718→            const m = nodeF.units[k];
   719→            if (m) {
   720→              m.position.y = bottomY(k) + slide;
   721→              m.visible = true;
   722→            }
   723→          }
   724→          // destination ghosts fill sequentially, growing upward from the base
   725→          for (let k = 0; k < count; k++) {
   726→            const gp = clamp01((tMs - 90 - k * UNIT_MS) / 110);
   727→            const s = Math.max(0.01, gp);
   728→            ghosts[k].scale.y = s;
   729→            ghosts[k].position.y = bottomY(dstLen + k) - HALF_H * (1 - s);
   730→          }
   731→        });
   732→
   733→        /* --- phase C: straighten + settle back onto its shelf --- */
   734→        await tweenMs(FLY_BACK, (p) => {
   735→          const e = p * p * (3 - 2 * p);
   736→          nodeF.group.quaternion.setFromAxisAngle(axis, TILT * (1 - e));
   737→          nodeF.group.position.y = THREE.MathUtils.lerp(basePos.y + POUR_LIFT, basePos.y, e);
   738→        });
   739→        nodeF.group.quaternion.identity();
   740→        nodeF.group.position.copy(basePos);
   741→        // pin every source unit back onto its exact stack slot
   742→        nodeF.units.forEach((m, k) => {
   743→          m.position.y = bottomY(k);
   744→          if (k < srcLen - count) {
   745→            m.scale.y = 1;
   746→            m.visible = true;
   747→          }
   748→        });
   749→
   750→        ghosts.forEach((m) => m.removeFromParent());
   751→        pouringRef.current = null;
   752→        // let the state reconciliation place the real destination units quietly
   753→        quietRef.current.add(to);
   754→      };
