    87→function bezierArc(a: THREE.Vector3, b: THREE.Vector3, apex: number, t: number, out: THREE.Vector3): THREE.Vector3 {
    88→  const midX = (a.x + b.x) / 2;
    89→  const midY = Math.max(a.y, b.y) + apex;
    90→  const midZ = (a.z + b.z) / 2;
    91→  const u = 1 - t;
    92→  out.set(
    93→    u * u * a.x + 2 * u * t * midX + t * t * b.x,
    94→    u * u * a.y + 2 * u * t * midY + t * t * b.y,
    95→    u * u * a.z + 2 * u * t * midZ + t * t * b.z,
    96→  );
    97→  return out;
    98→}

   250→    function buildBoard(state: number[][]) {
   251→      // clear previous tubes
   252→      for (const n of nodes) {
   253→        n.group.removeFromParent();
   254→        disposeObject(n.group);
   255→      }
   256→      nodes.length = 0;
   257→      scaleFxRef.current = [];
   258→      growRef.current = [];

   260→      // portrait screens: split into two depth rows so the board uses the
   261→      // vertical space instead of shrinking into a distant single row
   262→      const aspect = camera.aspect || (el.clientWidth || 1) / (el.clientHeight || 1);
   263→      const twoRows = aspect < 1.0 && state.length >= 4;
   264→      const rows = twoRows
   265→        ? [Math.ceil(state.length / 2), Math.floor(state.length / 2)]
   266→        : rowSplit(state.length);
   267→      const maxCount = Math.max(...rows);
   268→

   270→        const rowIdx = rows[0] && i < rows[0] ? 0 : 1;
   271→        const countInRow = rows[rowIdx];
   272→        const startInRow = rowIdx === 0 ? 0 : rows[0];
   273→        const k = i - startInRow;
   274→        const x = (k - (countInRow - 1) / 2) * SPACING;
   275→        const z = rowIdx === 0 ? -ROW_Z : ROW_Z;

   277→        const group = new THREE.Group();
   278→        group.position.set(x, 0, z);
   279→
   280→        const glass = new THREE.Mesh(glassGeo, glassMat);
   281→        glass.position.y = TUBE_H / 2 + 0.08;
   282→        group.add(glass);
   283→
   284→        const rim = new THREE.Mesh(new THREE.TorusGeometry(TUBE_R + 0.02, 0.045, 10, 26), rimMat);
   285→        rim.rotation.x = Math.PI / 2;
   286→        rim.position.y = TUBE_H + 0.08;
   287→        group.add(rim);
   288→
   289→        const ringMat = new THREE.MeshBasicMaterial({ color: 0xf5b301, transparent: true, opacity: 0 });
   290→        const ring = new THREE.Mesh(ringGeo, ringMat);
   291→        ring.rotation.x = Math.PI / 2;
   292→        ring.position.y = 0.09;
   293→        group.add(ring);
   294→
   295→        const hit = new THREE.Mesh(hitGeo, new THREE.MeshBasicMaterial({ transparent: true, opacity: 0, depthWrite: false }));
   296→        hit.position.y = TUBE_H / 2;
   297→        hit.userData.tubeIndex = i;
   298→        group.add(hit);
   299→
   300→        const liquid = new THREE.Group();

   303→        const node: TubeNode = { group, liquid, ring, ringMat, hit, baseX: x, baseZ: z, floatPhase: i * 0.83 + Math.abs(x) * 0.6, units: [] };

   305→        boardRoot.add(group);

   310→          m.scale.y = 0.01;

   312→          node.liquid.add(m);
   313→          growRef.current.push({ mesh: m, p: -k2 * 0.35 });
   314→        });
   315→      });
   316→
   317→      // camera fit
   318→      const width = (maxCount - 1) * SPACING + TUBE_R * 2 + 0.8;
   319→      const height = TUBE_H + 1.0;
   320→      const depth = (rows.length - 1) * ROW_Z * 2 + 1.4;
   321→      const fit = fitBoardCamera(camera, width, height, depth, new THREE.Vector3(0, height * 0.42, 0), {
   322→        headroom: 1.35,
   323→        margin: 1.04,
   324→        lookBias: -0.05,
   325→      });
   326→      lookTargetV.copy(fit.lookAt);
   327→    }

   329→    function syncTubes(state: number[][]) {
   330→      state.forEach((target, i) => {
   331→        const node = nodes[i];
   332→        if (!node) return;
   333→        const quiet = quietRef.current.has(i);
   334→        // shrink: remove extras
   335→        while (node.units.length > target.length) {
   336→          const m = node.units.pop();
   337→          if (m) m.removeFromParent();
   338→        }
   339→        // grow: add missing
   340→        while (node.units.length < target.length) {
   341→          const k = node.units.length;
   342→          const m = makeUnit(target[k], k);
   343→          if (quiet) {
   344→            m.scale.y = 1;
   345→          } else {
   346→            m.scale.y = 0.01;
   347→            growRef.current.push({ mesh: m, p: 0 });
   348→          }
   349→          node.units.push(m);
   350→          node.liquid.add(m);
   351→        }
   352→        // rebuild if colors differ with same length
   353→        if (node.units.length === target.length) {
   354→          const mismatch = target.some((c, k) => node.units[k].material !== liqMats[c % liqMats.length]);
   355→          if (mismatch) {
   356→            node.units.forEach((m) => m.removeFromParent());
   357→            node.units = [];
   358→            target.forEach((c, k) => {
   359→              const m = makeUnit(c, k);
   360→              node.units.push(m);
   361→              node.liquid.add(m);
   362→            });
   363→          }
   364→        }
   365→      });
   366→      quietRef.current.clear();
   367→    }

   369→    function burst(idx: number) {
   370→      const node = nodes[idx];
   371→      if (!node) return;
   372→      const n = 30;
   373→      const positions = new Float32Array(n * 3);
   374→      const vels = new Float32Array(n * 3);
   375→      const origin = new THREE.Vector3(node.baseX, TUBE_H * 0.62, node.baseZ);
   376→      for (let i = 0; i < n; i++) {
   377→        positions[i * 3] = origin.x;
   378→        positions[i * 3 + 1] = origin.y;
   379→        positions[i * 3 + 2] = origin.z;
   380→        const a = (i / n) * Math.PI * 2;
   381→        const r = 0.8 + Math.random() * 1.4;
   382→        vels[i * 3] = Math.cos(a) * r * 0.9;
   383→        vels[i * 3 + 1] = 1.6 + Math.random() * 1.8;
   384→        vels[i * 3 + 2] = Math.sin(a) * r * 0.55;
   385→      }
   386→      const geo = new THREE.BufferGeometry();
   387→      geo.setAttribute("position", new THREE.BufferAttribute(positions, 3));
   388→      const mat = new THREE.PointsMaterial({ color: 0xffd54f, size: 0.13, transparent: true, opacity: 1, depthWrite: false });
   389→      const points = new THREE.Points(geo, mat);
   390→      scene.add(points);
   391→      burstsRef.current.push({ points, mat, vels, born: performance.now(), life: 950 });
   392→    }

   435→    let running = true;
   436→    let elapsed = 0;
   437→    const lookTargetV = new THREE.Vector3(0, 1.2, 0);
   438→    const prevShkRef = { current: null as number | null };
   439→
   440→    const tick = () => {
   441→      if (!running) return;
   442→      raf = requestAnimationFrame(tick);
   443→      const dt = Math.min(clock.getDelta(), 0.05);
   444→      elapsed += dt;
   445→      const now = performance.now();
   446→      const { selected: sel, hinted: hin, shakeIdx: shk } = propsRef.current;
   447→
   448→      // unit grow pop (easeOutBack)
   449→      growRef.current = growRef.current.filter((g) => {
   450→        g.p += dt * 3.4;
   451→        const p = clamp01(g.p);
   452→        const back = 1 + 2.2 * Math.pow(p - 1, 3) + 1.2 * Math.pow(p - 1, 2);
   453→        g.mesh.scale.y = Math.max(0.01, p < 1 ? back : 1);
   454→        if (g.p >= 1) {
   455→          g.mesh.scale.y = 1;
   456→          return false;
   457→        }
   458→        return true;
   459→      });
   460→
   461→      // scale fx (source shrink handled by pour tween, kept generic)
   462→      scaleFxRef.current = scaleFxRef.current.filter((fx) => {
   463→        const p = clamp01((now - fx.t0) / fx.dur);
   464→        fx.mesh.scale.y = fx.from + (fx.to - fx.from) * p;
   465→        return p < 1;
   466→      });
   467→
   468→      // per-tube visuals
   469→      nodes.forEach((node, i) => {
   470→        if (pouringRef.current !== i) {
   471→          // gentle idle bob so the tubes float in the air
   472→          const bob = Math.sin(elapsed * 1.15 + node.floatPhase) * 0.07;
   473→          const targetLift = (sel === i ? 0.42 : 0) + bob;
   474→          node.group.position.y += (targetLift - node.group.position.y) * Math.min(1, dt * 8);
   475→          // shake (only re-armed when the shake target changes)
   476→          if (shk !== prevShkRef.current) {
   477→            prevShkRef.current = shk;
   478→            if (shk === i) shakeRef.current = { idx: i, until: now + 380 };
   479→          }
   480→          if (shakeRef.current && shakeRef.current.idx === i) {
   481→            const remain = shakeRef.current.until - now;
   482→            if (remain <= 0) {
   483→              shakeRef.current = null;
   484→              node.group.position.x = node.baseX;
   485→            } else {
   486→              const decay = remain / 380;
   487→              node.group.position.x = node.baseX + Math.sin(elapsed * 55) * 0.09 * decay;
   488→            }
   489→          } else {
   490→            node.group.position.x += (node.baseX - node.group.position.x) * Math.min(1, dt * 10);
   491→          }
   492→        }
   493→
   494→        // ring state
   495→        const complete = completeRef.current[i] ?? false;
   496→        const isSel = sel === i;
   497→        const isHint = hin !== null && (hin.from === i || hin.to === i);
   498→        if (isSel) {
   499→          node.ringMat.color.setHex(0xf5a623);
   500→          node.ringMat.opacity = 0.7 + Math.sin(elapsed * 7) * 0.25;
   501→          node.ring.scale.setScalar(1 + Math.sin(elapsed * 7) * 0.05);
   502→        } else if (isHint) {
   503→          node.ringMat.color.setHex(0x2dd4a0);
   504→          node.ringMat.opacity = 0.55 + Math.sin(elapsed * 6) * 0.35;
   505→          node.ring.scale.setScalar(1 + Math.sin(elapsed * 6) * 0.07);
   506→        } else if (complete) {
   507→          node.ringMat.color.setHex(0xffd54f);
   508→          node.ringMat.opacity = 0.42 + Math.sin(elapsed * 2.4) * 0.18;
   509→          node.ring.scale.setScalar(1);
   510→        } else {
   511→          node.ringMat.opacity = 0;
   512→        }
   513→      });
   514→
   515→      // sparkle bursts
   516→      burstsRef.current = burstsRef.current.filter((b) => {
   517→        const age = now - b.born;
   518→        if (age > b.life) {
   519→          b.points.removeFromParent();
   520→          b.points.geometry.dispose();
   521→          b.mat.dispose();
   522→          return false;
   523→        }

   614→          nodeF.group.position.z = THREE.MathUtils.lerp(basePos.z, nodeT.baseZ, e);

   616→        });

   651→          // source units drain sequentially

   669→          nodeF.group.position.z = THREE.MathUtils.lerp(nodeT.baseZ, basePos.z, e);
   670→          nodeF.group.position.y = THREE.MathUtils.lerp(basePos.y + liftY, basePos.y, e);
   671→        });
